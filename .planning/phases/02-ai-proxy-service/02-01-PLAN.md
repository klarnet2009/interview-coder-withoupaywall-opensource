    ```typescript
    import { config } from '../config'
    import { ProcessingProvider, ProcessingResult, ApiProvider, ... } from './types'
    import { OpenAIProcessingProvider } from './providers/openai.provider'
    import { GeminiProcessingProvider } from './providers/gemini.provider'
    import { AnthropicProcessingProvider } from './providers/anthropic.provider'

    class ProcessingService {
      private providers: Map<ApiProvider, ProcessingProvider>

      constructor() {
        this.providers = new Map()
        // Initialize providers based on available API keys
        if (config.OPENAI_API_KEY) {
          this.providers.set('openai', new OpenAIProcessingProvider(config.OPENAI_API_KEY))
        }
        if (config.GEMINI_API_KEY) {
          this.providers.set('gemini', new GeminiProcessingProvider(config.GEMINI_API_KEY))
        }
        if (config.ANTHROPIC_API_KEY) {
          this.providers.set('anthropic', new AnthropicProcessingProvider(config.ANTHROPIC_API_KEY))
        }
      }

      getProvider(provider: ApiProvider): ProcessingProvider {
        const p = this.providers.get(provider)
        if (!p) {
          if (!['openai', 'gemini', 'anthropic'].includes(provider)) {
            throw new Error(`Unknown provider: ${provider}`)
          }
          throw new Error(`Provider ${provider} is not configured. Set the ${provider.toUpperCase()}_API_KEY environment variable.`)
        }
        return p
      }

      async extractProblem(provider: ApiProvider, request: ExtractProblemRequest): ProcessingResult<ProblemInfo> {
        try {
          const p = this.getProvider(provider)
          return await p.extractProblem(request)
        } catch (err) {
          return { success: false, error: err instanceof Error ? err.message : 'Unknown error', statusCode: 400 }
        }
      }

      async generateSolution(provider: ApiProvider, request: GenerateSolutionRequest): ProcessingResult<string> {
        // Similar delegation pattern
      }

      async generateDebug(provider: ApiProvider, request: GenerateDebugRequest): ProcessingResult<string> {
        // Similar delegation pattern
      }
    }

    export const processingService = new ProcessingService()
    ```
